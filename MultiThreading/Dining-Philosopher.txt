// solution 1 using locks

import java.util.concurrent.locks.ReentrantLock;

class DiningPhilosophers {

    private final ReentrantLock[] forks;

    public DiningPhilosophers() {
        forks = new ReentrantLock[5];
        for (int i = 0; i < 5; i++) {
            forks[i] = new ReentrantLock();
        }
    }

    public void wantsToEat(
        int philosopher,
        Runnable pickLeftFork,
        Runnable pickRightFork,
        Runnable eat,
        Runnable putLeftFork,
        Runnable putRightFork
    ) throws InterruptedException {

        int left = philosopher;
        int right = (philosopher+1)%5;

        int mini = Math.min(left, right);
        int maxi = Math.max(left, right);

        forks[mini].lockInterruptibly();

        try{
            forks[maxi].lockInterruptibly();
            try{
                pickLeftFork.run();
                pickRightFork.run();

                eat.run();

                putLeftFork.run();
                putRightFork.run();
            } finally {
                forks[maxi].unlock();
            }
        } finally {
            forks[mini].unlock();
        }
    }
}

lock has 2 methods -> lock and lockInterruptibly , lock waits even if someone calls interrupt but lockInterruptibly stops the waiting if some calls interrupt.

import java.util.concurrent.locks.ReentrantLock;

class DiningPhilosophers {

    private Semaphore[] forks;
    private Semaphore rooms = new Semaphore(4);

    public DiningPhilosophers() {
        forks = new Semaphore[5];
        for (int i = 0; i < 5; i++) {
            forks[i] = new Semaphore(1);
        }
    }

    public void wantsToEat(
        int philosopher,
        Runnable pickLeftFork,
        Runnable pickRightFork,
        Runnable eat,
        Runnable putLeftFork,
        Runnable putRightFork
    ) throws InterruptedException {

        int left = philosopher;
        int right = (philosopher+1)%5;
        rooms.acquire();

        try {
            forks[left].acquire();
            try {
                forks[right].acquire();
                try {
                    pickLeftFork.run();
                    pickRightFork.run();

                    eat.run();

                    putLeftFork.run();
                    putRightFork.run();
                } finally {
                    forks[right].release();
                } 
            }finally {
                forks[left].release();
            }
        } finally {
            rooms.release();
        }
    }
}

class DiningPhilosophers {

    private static final int THINKING = 0;
    private static final int HUNGRY = 1;
    private static final int EATING = 2;

    private final int[] state = new int[5];

    private final Object monitor = new Object();

    public DiningPhilosophers() {
    }

    private boolean canEat(int i) {

        int left = (i + 4) % 5;
        int right = (i + 1) % 5;

        return state[i] == HUNGRY &&
               state[left] != EATING &&
               state[right] != EATING;
    }

    public void wantsToEat(
        int philosopher,
        Runnable pickLeftFork,
        Runnable pickRightFork,
        Runnable eat,
        Runnable putLeftFork,
        Runnable putRightFork
    ) throws InterruptedException {

        boolean acquired = false;

        try {

            synchronized (monitor) {

                state[philosopher] = HUNGRY;

                while (!canEat(philosopher)) {
                    monitor.wait();
                }

                state[philosopher] = EATING;
                acquired = true;
            }

            pickLeftFork.run();
            pickRightFork.run();

            try {
                eat.run();
            } finally {
                putLeftFork.run();
                putRightFork.run();
            }

        } finally {

            synchronized (monitor) {

                if (!acquired || state[philosopher] == EATING) {
                    state[philosopher] = THINKING;
                }

                monitor.notifyAll();
            }
        }
    }
}
